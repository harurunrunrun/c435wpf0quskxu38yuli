## 使う場面

定数項 0 の $F$ に対し、$\exp F$ の係数を多数求めます。互いに区別される部品を集合にまとめる数え上げなどで、和の情報を積の情報に戻すと、この形が出ます。

定義の $\sum F^k/k!$ を冪ごとに作る代わりに、既に正しい係数を保ったまま精度を伸ばします。

## 考え方

先頭 $m$ 項が正しい $G$ を用意し、対数を先頭 $2m$ 項まで求めます。補正は

$$G_{\mathrm{new}}=G(1+F-\log G)\pmod{x^{2m}}$$

です。初期値は $G=1$。$F-\log G$ は低い次数では 0 なので、新しい次数だけを直します。$F=x$ なら、$G$ は $1$、$1+x$、$1+x+x^2/2+x^3/6$ と伸びます。

## 適用条件

$F(0)=0$、$G(0)=1$ とします。$F$ の定数項が 0 でなければ、この形式的指数の定義は使えません。掲載実装は法 998244353、入力係数 $[0,p)$、$0\le N\le2^{21}$ です。分母に使う $1,\ldots,N-1$ が可逆になる精度に限ります。

入力が空なら零級数として扱います。必要な次数以降の係数を知らなくても、先頭 $N$ 項を作るのに影響はありません。

## なぜ正しいか

$\Delta=F-\log G$ と置きます。今の $G$ が先頭 $m$ 項まで正しければ、$\Delta$ は $x^m$ の倍数です。そのため $\Delta^2$ は $x^{2m}$ の倍数になり、

$$\log(1+\Delta)\equiv\Delta\pmod{x^{2m}}$$

となります。更新後の対数は $\log G+\log(1+\Delta)\equiv F$ です。定数項 1 の列について指数と対数は逆になり、必要な長さの指数が得られます。

## 計算量

$G'=F'G$ の係数を一項ずつ解く方法は時間 $O(N^2)$ です。長さを倍にする更新では、各段階の対数・逆数・積が $O(m\log m)$ なので、その幾何級数の総和は $O(N\log N)$ です。空間は $O(N)$。この実装でも、更新のたびに対数を最初から計算してよく、次数の和が倍々に増えるため計算量は保たれます。

## 実装

`fps_exponential(f,n)` は `n` 項を返します。コードには対数・逆数・畳み込みも含めています。最後の更新は 2 の冪に切り上げず、必要な長さで止めます。

```cpp
#include <bits/stdc++.h>
using namespace std;
namespace ntt998 {
constexpr int MOD=998244353;
int power(int a,int n) {
    int r=1; for(;n;n>>=1,a=1LL*a*a%MOD) if(n&1) r=1LL*r*a%MOD; return r;
}
void transform(vector<int>& a,bool inverse) {
    int n=a.size(); assert(n>0 && (n&(n-1))==0 && n<=(1<<23));
    for(int i=1,j=0;i<n;++i) {
        int bit=n>>1; for(;j&bit;bit>>=1) j^=bit; j^=bit;
        if(i<j) swap(a[i],a[j]);
    }
    for(int len=2;len<=n;len<<=1) {
        int root=power(3,(MOD-1)/len); if(inverse) root=power(root,MOD-2);
        for(int i=0;i<n;i+=len) {
            int w=1;
            for(int j=0;j<len/2;++j) {
                int u=a[i+j],v=1LL*a[i+j+len/2]*w%MOD;
                a[i+j]=u+v<MOD?u+v:u+v-MOD;
                a[i+j+len/2]=u-v>=0?u-v:u-v+MOD; w=1LL*w*root%MOD;
            }
        }
    }
    if(inverse) { int inv=power(n,MOD-2); for(int& x:a) x=1LL*x*inv%MOD; }
}
vector<int> convolution(vector<int> a,vector<int> b) {
    if(a.empty() || b.empty()) return {};
    size_t need=a.size()+b.size()-1; assert(need<=size_t(1<<23));
    int n=1; while(size_t(n)<need) n<<=1;
    for(int& x:a) x=(x%MOD+MOD)%MOD;
    for(int& x:b) x=(x%MOD+MOD)%MOD;
    a.resize(n); b.resize(n); transform(a,false); transform(b,false);
    for(int i=0;i<n;++i) a[i]=1LL*a[i]*b[i]%MOD;
    transform(a,true); a.resize(need); return a;
}
}

vector<int> fps_inverse(const vector<int>& f,int n) {
    assert(0<=n && n<=(1<<21));
    if(n==0) return {};
    assert(!f.empty() && f[0]!=0);
    vector<int> g{ntt998::power(f[0],ntt998::MOD-2)};
    while(int(g.size())<n) {
        int m=min(n,2*int(g.size()));
        vector<int> part(f.begin(),f.begin()+min(m,int(f.size())));
        auto correction=ntt998::convolution(part,g);
        correction.resize(m);
        for(int& x:correction) x=(ntt998::MOD-x)%ntt998::MOD;
        correction[0]=(correction[0]+2)%ntt998::MOD;
        g=ntt998::convolution(g,correction);
        g.resize(m);
    }
    return g;
}

vector<int> fps_logarithm(const vector<int>& f,int n) {
    assert(0<=n && n<=(1<<21));
    if(n==0) return {};
    assert(!f.empty() && f[0]==1);
    vector<int> derivative(max(0,min(n,int(f.size()))-1));
    for(int i=1;i<=int(derivative.size());++i)
        derivative[i-1]=1LL*i*f[i]%ntt998::MOD;
    auto quotient=ntt998::convolution(derivative,fps_inverse(f,n));
    quotient.resize(n-1);
    vector<int> result(n),inverse(n);
    if(n>1) inverse[1]=1;
    for(int i=2;i<n;++i)
        inverse[i]=ntt998::MOD-1LL*(ntt998::MOD/i)*inverse[ntt998::MOD%i]%ntt998::MOD;
    for(int i=1;i<n;++i) result[i]=1LL*quotient[i-1]*inverse[i]%ntt998::MOD;
    return result;
}

vector<int> fps_exponential(const vector<int>& f,int n) {
    assert(0<=n && n<=(1<<21));
    if(n==0) return {};
    assert(f.empty() || f[0]==0);
    vector<int> g{1};
    while(int(g.size())<n) {
        int m=min(n,2*int(g.size()));
        auto delta=fps_logarithm(g,m);
        for(int i=0;i<m;++i) {
            int value=i<int(f.size())?f[i]:0;
            delta[i]=(value-delta[i]+ntt998::MOD)%ntt998::MOD;
        }
        delta[0]=(delta[0]+1)%ntt998::MOD;
        g=ntt998::convolution(g,delta);
        g.resize(m);
    }
    return g;
}
```
